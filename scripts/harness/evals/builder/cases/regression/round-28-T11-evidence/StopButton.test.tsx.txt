import { describe, it, expect, vi, beforeEach } from 'vitest';
import { render, screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { StopButton } from './StopButton';
import { useIncidentStore } from '@store/useIncidentStore';

vi.mock('react-i18next', () => ({
  useTranslation: () => ({
    t: (_key: string, defaultVal: string) => defaultVal,
  }),
}));

vi.mock('@store/useIncidentStore', () => ({
  useIncidentStore: vi.fn(),
}));

describe('StopButton', () => {
  const mockStopIncident = vi.fn().mockResolvedValue(undefined);

  beforeEach(() => {
    vi.clearAllMocks();
    (useIncidentStore as unknown as ReturnType<typeof vi.fn>).mockReturnValue({
      activeIncident: null,
      isLoading: false,
      error: null,
      startIncident: vi.fn(),
      stopIncident: mockStopIncident,
      hydrateActiveIncident: vi.fn(),
    });
  });

  it('renders with aria-label matching the i18n stop value (BR-12)', () => {
    render(<StopButton />);
    expect(screen.getByRole('button', { name: 'STOP' })).toBeInTheDocument();
  });

  it('aria-label attribute equals the translated stop label', () => {
    render(<StopButton />);
    expect(screen.getByRole('button', { name: 'STOP' })).toHaveAttribute(
      'aria-label',
      'STOP'
    );
  });

  it('fires stopIncident store action on tap (BR-13)', async () => {
    const user = userEvent.setup();
    render(<StopButton />);
    await user.click(screen.getByRole('button', { name: 'STOP' }));
    expect(mockStopIncident).toHaveBeenCalledOnce();
  });
});
